cmake_minimum_required(VERSION 3.22)
project(DyneImGuiConsumer LANGUAGES CXX)
set(CMAKE_FIND_LIBRARY_SUFFIXES .a)
include(CheckPIESupported)
check_pie_supported()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(imgui 1.93 CONFIG REQUIRED)
add_executable(imgui main.cpp)
target_link_libraries(imgui PRIVATE imgui::imgui)
find_package(PkgConfig REQUIRED)
list(APPEND PKG_CONFIG_ARGN --static)
pkg_check_modules(IMGUI REQUIRED IMPORTED_TARGET imgui)
add_executable(imgui-pkgconfig main.cpp)
target_link_libraries(imgui-pkgconfig PRIVATE PkgConfig::IMGUI)
enable_testing()
foreach(consumer imgui imgui-pkgconfig)
  add_test(NAME ${consumer} COMMAND ${consumer})
  set_tests_properties(${consumer} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
endforeach()
