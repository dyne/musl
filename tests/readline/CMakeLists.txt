cmake_minimum_required(VERSION 3.22)
project(DyneReadlineConsumer LANGUAGES C)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/RequireStaticPIE.cmake")
dyne_require_static_pie()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(PkgConfig REQUIRED)
list(APPEND PKG_CONFIG_ARGN --static)
pkg_check_modules(READLINE REQUIRED IMPORTED_TARGET readline)
pkg_check_modules(HISTORY REQUIRED IMPORTED_TARGET history)
add_executable(readline main.c)
target_link_libraries(readline PRIVATE PkgConfig::READLINE)
add_executable(history history.c)
target_link_libraries(history PRIVATE PkgConfig::HISTORY)
enable_testing()
add_test(NAME readline COMMAND readline)
add_test(NAME history COMMAND history)
set_tests_properties(readline history PROPERTIES TIMEOUT 10
  ENVIRONMENT "INPUTRC=/dev/null;TERM=dumb")
